export const EnergyBars = ({ data }) => {
  // compile() renders server-side HTML without hydration, so use pure SVG.
  const width = 730;
  const height = 250;
  const left = 54;
  const top = 14;
  const bottom = 32;
  const plotWidth = width - left - 12;
  const plotHeight = height - top - bottom;
  const maxTotal = Math.max(
    1,
    ...data.map((day) => day.electricity + day.gas + day.oil),
  );
  const barWidth = Math.min(15, (plotWidth / data.length) * 0.7);
  const colors = { electricity: "#fcd656", gas: "#ec6666", oil: "#1479d4" };

  return (
    <svg
      role="img"
      aria-label="Consommation quotidienne par énergie"
      viewBox={`0 0 ${width} ${height}`}
      width={width}
      height={height}
      style={{ maxWidth: "100%", height: "auto" }}
    >
      <title>Consommation quotidienne par énergie</title>
      {[0, 0.5, 1].map((fraction) => {
        const y = top + plotHeight * (1 - fraction);
        return (
          <g key={fraction}>
            <line x1={left} y1={y} x2={width - 12} y2={y} stroke="#cbd5e1" />
            <text
              x={left - 8}
              y={y + 4}
              textAnchor="end"
              fontSize="10"
              fill="#64748b"
            >
              {Math.round(maxTotal * fraction).toLocaleString("fr-FR")}
            </text>
          </g>
        );
      })}
      {data.map((day, index) => {
        const x = left + ((index + 0.5) * plotWidth) / data.length;
        let sum = 0;
        return (
          <g
            key={day.date}
            data-energy-day={new Date(day.date).toISOString().slice(0, 10)}
          >
            {Object.entries(colors).map(([fuel, color]) => {
              const amount = day[fuel];
              const rectangleHeight = (amount / maxTotal) * plotHeight;
              sum += amount;
              return (
                <rect
                  key={fuel}
                  x={x - barWidth / 2}
                  y={top + plotHeight - (sum / maxTotal) * plotHeight}
                  width={barWidth}
                  height={rectangleHeight}
                  fill={color}
                />
              );
            })}
            {index % 7 === 0 ? (
              <text
                x={x}
                y={height - 10}
                textAnchor="middle"
                fontSize="10"
                fill="#64748b"
              >
                {new Date(day.date).getUTCDate()}/01
              </text>
            ) : null}
          </g>
        );
      })}
    </svg>
  );
};

export const EnergyHalfPie = ({ data }) => {
  const total = Math.max(
    1,
    data.reduce((sum, entry) => sum + entry.value, 0),
  );
  const colors = ["#fcd656", "#ec6666", "#1479d4"];
  const center = { x: 150, y: 140 };
  const radius = 110;
  const point = (angle) => ({
    x: center.x + radius * Math.cos((angle * Math.PI) / 180),
    y: center.y - radius * Math.sin((angle * Math.PI) / 180),
  });

  return (
    <svg
      role="img"
      aria-label="Répartition de la consommation énergétique"
      viewBox="0 0 300 160"
      width="300"
      height="160"
      style={{ maxWidth: "100%", height: "auto" }}
    >
      <title>Répartition de la consommation énergétique</title>
      {data.map((entry, index) => {
        const before = data
          .slice(0, index)
          .reduce((sum, item) => sum + item.value, 0);
        const start = point(180 * (1 - before / total));
        const end = point(180 * (1 - (before + entry.value) / total));
        const largeArc = entry.value / total > 0.5 ? 1 : 0;
        return (
          <path
            key={entry.name}
            d={`M ${center.x} ${center.y} L ${start.x} ${start.y} A ${radius} ${radius} 0 ${largeArc} 1 ${end.x} ${end.y} Z`}
            fill={colors[index % colors.length]}
            stroke="white"
            strokeWidth="1"
          />
        );
      })}
    </svg>
  );
};
