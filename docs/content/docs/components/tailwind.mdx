#### Preview

Use a simple Tailwind tag to support Tailwind in your document.

<Frame background="subtle"><PreviewImage src="/docs/images/previews/tailwind-06135a78/document.jpg" /></Frame>

<div style={{paddingTop: "1rem", paddingBottom: "1rem"}}><CodeBlocks>
<CodeBlock title="template.tsx">
```tsx
import React from "react";
import { Tailwind } from "react-print-pdf";

export const Document = () => (
  <Tailwind>
    <div className="rounded-2xl bg-gradient-to-tr from-blue-500 to-blue-700 p-12" />
    <p className="py-12 text-slate-800">
      This is a Tailwind component. All children of this component will have
      access to the Tailwind CSS classes.
    </p>
  </Tailwind>
);

```
</CodeBlock>
<CodeBlock title="styles.css">
```css
@import url("https://fonts.googleapis.com/css2?family=Inter:wght@400;700&display=swap");

html,
body {
  font-size: 28px;
  font-family: "Inter", sans-serif;
}

@page {
  size: A4;
}

```
</CodeBlock>
</CodeBlocks></div>

### Examples

#### Tailwind v4 CSS configuration

Use Tailwind v4 directives such as `@theme`, `@utility`, and `@custom-variant` through the stylesheet prop.

<Frame background="subtle"><PreviewImage src="/docs/images/previews/tailwind-5d7ee90c/document.jpg" /></Frame>

<div style={{paddingTop: "1rem", paddingBottom: "1rem"}}><CodeBlocks>
<CodeBlock title="template.tsx">
```tsx
import React from "react";
import { Tailwind } from "react-print-pdf";

export const Document = () => (
  <Tailwind
    stylesheet={`@theme {
  --color-primary: #6484cf;
}`}
  >
    <div className="rounded-2xl bg-primary p-12" />
    <p className="py-12 text-slate-800">
      This component uses a CSS-first Tailwind v4 theme token.
    </p>
  </Tailwind>
);

```
</CodeBlock>
<CodeBlock title="styles.css">
```css
@import url("https://fonts.googleapis.com/css2?family=Inter:wght@400;700&display=swap");

html,
body {
  font-size: 28px;
  font-family: "Inter", sans-serif;
}

@page {
  size: A4;
}

```
</CodeBlock>
</CodeBlocks></div>

#### Legacy JavaScript config

Existing JavaScript theme configuration remains supported through Tailwind v4's compatibility layer.

<Frame background="subtle"><PreviewImage src="/docs/images/previews/tailwind-75ebb237/document.jpg" /></Frame>

<div style={{paddingTop: "1rem", paddingBottom: "1rem"}}><CodeBlocks>
<CodeBlock title="template.tsx">
```tsx
import React from "react";
import { Tailwind } from "react-print-pdf";

export const Document = () => (
  <Tailwind
    config={{
      theme: {
        extend: {
          colors: {
            primary: "#6484cf",
          },
        },
      },
    }}
  >
    <div className="rounded-2xl bg-primary p-12" />
  </Tailwind>
);

```
</CodeBlock>

```css
@page {
  size: A4;

```
</CodeBlock>
</CodeBlocks></div>

#### Disable Preflight

Set `preflight={false}` to omit Tailwind's Preflight reset.

<Frame background="subtle"><PreviewImage src="/docs/images/previews/tailwind-b289d690/document.jpg" /></Frame>

<div style={{paddingTop: "1rem", paddingBottom: "1rem"}}><CodeBlocks>
<CodeBlock title="template.tsx">
```tsx
import React from "react";
import { Tailwind } from "react-print-pdf";

export const Document = () => (
  <Tailwind preflight={false}>
    <h1 className="text-2xl font-bold">Level 1 Header</h1>
    <p className="text-slate-800">
      Tailwind utilities remain available without the reset.
    </p>
  </Tailwind>
);

```
</CodeBlock>
<CodeBlock title="styles.css">
```css
@import url("https://fonts.googleapis.com/css2?family=Inter:wght@400;700&display=swap");

html,
body {
  font-size: 28px;
  font-family: "Inter", sans-serif;
}

@page {
  size: A4;
}

```
</CodeBlock>
</CodeBlocks></div>
